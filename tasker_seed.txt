CREATE DATABASE Tasker;
USE Tasker;

CREATE TABLE users(
id INT(6) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(100) NOT NULL,
email VARCHAR(100) NOT NULL,
password VARCHAR(100) NOT NULL  
);

CREATE TABLE rooms(
id INT(6) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
title VARCHAR(100) NOT NULL,
deleted INT(6) NOT NULL DEFAULT 0,
creator_id INT(6) NOT NULL
);

CREATE TABLE boards(
id INT(6) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
title VARCHAR(100) NOT NULL,
room_id INT(6) NOT NULL 
);

CREATE TABLE tasks(
id INT(6) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
title VARCHAR(100) NOT NULL,
description VARCHAR(500) NOT NULL,
stage INT(1) NOT NULL,
executor_id INT(6),
board_id INT(6) NOT NULL
);

CREATE TABLE accounting(
id INT(6) UNSIGNED AUTO_INCREMENT PRIMARY KEY,
user_id INT(6) NOT NULL,
room_id INT(6) NOT NULL
);

ALTER TABLE rooms MODIFY COLUMN creator_id INT(6) UNSIGNED;

ALTER TABLE boards ADD FOREIGN KEY (room_id) REFERENCES rooms(id);

ALTER TABLE tasks MODIFY COLUMN executor_id INT(6) UNSIGNED;
ALTER TABLE tasks ADD FOREIGN KEY (executor_id) REFERENCES users(id);

ALTER TABLE tasks MODIFY COLUMN board_id INT(6) UNSIGNED;
ALTER TABLE tasks ADD FOREIGN KEY (board_id) REFERENCES boards(id);

ALTER TABLE accounting MODIFY COLUMN user_id INT(6) UNSIGNED;
ALTER TABLE accounting ADD FOREIGN KEY (user_id) REFERENCES users(id);

ALTER TABLE accounting MODIFY COLUMN room_id INT(6) UNSIGNED;
